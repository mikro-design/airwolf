# A separate native receive module. No GNU Radio, PMT, VOLK or Boost dependency.
enable_language(C)
set(AIRWOLF_LORA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../lora-deps/gr-lora_sdr-862746dd1cf635c9c8a4bfbaa2c3a0ec3a5306c9" CACHE PATH "Pinned, patched LoRa receive source")
if(NOT EXISTS "${AIRWOLF_LORA_SOURCE}/lib/frame_sync_impl.cc")
  message(FATAL_ERROR "Run node scripts/prepare-lora.mjs before enabling LoRa reception")
endif()
set(lora_blocks frame_sync fft_demod gray_mapping deinterleaver hamming_dec header_decoder dewhitening crc_verif)
set(lora_sources)
foreach(block IN LISTS lora_blocks)
  list(APPEND lora_sources "${AIRWOLF_LORA_SOURCE}/lib/${block}_impl.cc")
endforeach()
add_library(airwolf_lora_phy STATIC ${lora_sources} "${AIRWOLF_LORA_SOURCE}/lib/kiss_fft.c")
target_include_directories(airwolf_lora_phy PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include" "${AIRWOLF_LORA_SOURCE}/include" "${AIRWOLF_LORA_SOURCE}/lib")
if(MSVC)
  target_compile_options(airwolf_lora_phy PRIVATE /w)
  target_compile_definitions(airwolf_lora_phy PRIVATE _USE_MATH_DEFINES)
else()
  target_compile_options(airwolf_lora_phy PRIVATE -w)
endif()
add_library(airwolf_lora_receiver STATIC receiver.cpp)
target_include_directories(airwolf_lora_receiver PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include" "${AIRWOLF_LORA_SOURCE}/include" "${AIRWOLF_LORA_SOURCE}/lib")
target_link_libraries(airwolf_lora_receiver PUBLIC airwolf_dsp PRIVATE airwolf_lora_phy)
if(MSVC)
  target_compile_definitions(airwolf_lora_receiver PRIVATE _USE_MATH_DEFINES)
endif()
add_executable(airwolf-lora-feed tests/receive_feed.cpp)
target_link_libraries(airwolf-lora-feed PRIVATE airwolf_lora_receiver)
add_executable(airwolf-lora-tests tests/core_test.cpp)
target_include_directories(airwolf-lora-tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(airwolf-lora-tests PRIVATE airwolf_lora_receiver)
add_test(NAME native-lora-core COMMAND airwolf-lora-tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/hackrf-zero4.ci8")
if(AIRWOLF_SIGNAL_DEPS EQUAL 0)
  add_test(NAME native-lora-independent COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/independent.py" "$<TARGET_FILE:airwolf-lora-feed>")
  set_tests_properties(native-lora-independent PROPERTIES TIMEOUT 120)
  add_test(NAME native-lora-replay COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/replay.py" "$<TARGET_FILE:airwolf-lora-feed>")
  set_tests_properties(native-lora-replay PROPERTIES TIMEOUT 180)
endif()
if(Python3_Interpreter_FOUND)
  add_test(NAME native-lora-recorded COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/recorded.py" "$<TARGET_FILE:airwolf-lora-feed>")
  set_tests_properties(native-lora-recorded PROPERTIES TIMEOUT 60)
endif()
# Opt-in performance evidence, deliberately not a timing-sensitive CTest gate.
add_executable(airwolf-lora-benchmark tests/benchmark.cpp)
target_link_libraries(airwolf-lora-benchmark PRIVATE airwolf_lora_receiver)

if(AIRWOLF_WITH_HACKRF)
  add_executable(airwolf-lora-radio ../src/main.cpp ../src/iq_decode.cpp)
  target_compile_definitions(airwolf-lora-radio PRIVATE AIRWOLF_LORA)
  target_link_libraries(airwolf-lora-radio PRIVATE airwolf_lora_receiver ${HACKRF_TARGET} Threads::Threads)
  if(WIN32)
    target_link_libraries(airwolf-lora-radio PRIVATE winmm)
  endif()
  set_target_properties(airwolf-lora-radio PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(airwolf-lora-radio PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
  endforeach()
endif()
