if(AIRWOLF_WITH_HACKRF)
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(HACKRF QUIET libhackrf)
  endif()
  # Some distribution .pc files contain stale/private libusb include paths.
  # Our public API dependency is hackrf.h and the shared libhackrf only.
  # Use pkg-config as search hints, validating the paths we actually consume.
  find_path(HACKRF_INCLUDE_DIR NAMES hackrf.h HINTS ${HACKRF_INCLUDE_DIRS} PATH_SUFFIXES libhackrf REQUIRED)
  find_library(HACKRF_LIBRARY NAMES hackrf libhackrf HINTS ${HACKRF_LIBRARY_DIRS} REQUIRED)
  add_library(hackrf_external UNKNOWN IMPORTED)
  set_target_properties(hackrf_external PROPERTIES IMPORTED_LOCATION "${HACKRF_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${HACKRF_INCLUDE_DIR}")
  set(HACKRF_TARGET hackrf_external)
  add_executable(airwolf-radio src/main.cpp)
  target_link_libraries(airwolf-radio PRIVATE airwolf_dsp ${HACKRF_TARGET} Threads::Threads)
  if(WIN32)
    target_link_libraries(airwolf-radio PRIVATE winmm)
  endif()
  set_target_properties(airwolf-radio PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(airwolf-radio PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
  endforeach()
  install(TARGETS airwolf-radio RUNTIME DESTINATION bin)
endif()
