cmake_minimum_required(VERSION 3.20)
if(CMAKE_HOST_APPLE AND NOT CMAKE_OSX_DEPLOYMENT_TARGET)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "15.0" CACHE STRING "Minimum supported macOS" FORCE)
endif()
project(airwolf_radio VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/W4)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
find_package(Threads REQUIRED)
option(AIRWOLF_WITH_HACKRF "Build the direct libhackrf receiver" ON)
add_library(airwolf_dsp INTERFACE)
target_include_directories(airwolf_dsp INTERFACE src)
add_executable(airwolf-native-tests tests/core_test.cpp)
target_link_libraries(airwolf-native-tests PRIVATE airwolf_dsp Threads::Threads)
if(AIRWOLF_WITH_HACKRF)
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(HACKRF QUIET IMPORTED_TARGET libhackrf)
  endif()
  if(TARGET PkgConfig::HACKRF)
    set(HACKRF_TARGET PkgConfig::HACKRF)
  else()
    find_path(HACKRF_INCLUDE_DIR NAMES hackrf.h PATH_SUFFIXES libhackrf REQUIRED)
    find_library(HACKRF_LIBRARY NAMES hackrf libhackrf REQUIRED)
    add_library(hackrf_external UNKNOWN IMPORTED)
    set_target_properties(hackrf_external PROPERTIES IMPORTED_LOCATION "${HACKRF_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${HACKRF_INCLUDE_DIR}")
    set(HACKRF_TARGET hackrf_external)
  endif()
  add_executable(airwolf-radio src/main.cpp)
  target_link_libraries(airwolf-radio PRIVATE airwolf_dsp ${HACKRF_TARGET} Threads::Threads)
  set_target_properties(airwolf-radio PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(airwolf-radio PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
  endforeach()
  install(TARGETS airwolf-radio RUNTIME DESTINATION bin)
endif()
enable_testing()
add_test(NAME native-core COMMAND airwolf-native-tests)
add_executable(airwolf-packet-tests tests/packet_test.cpp)
target_link_libraries(airwolf-packet-tests PRIVATE airwolf_dsp Threads::Threads)
add_test(NAME native-packets COMMAND airwolf-packet-tests)
add_executable(airwolf-ble-phy-tests tests/ble_phy_test.cpp)
target_link_libraries(airwolf-ble-phy-tests PRIVATE airwolf_dsp Threads::Threads)
add_test(NAME native-ble-phys COMMAND airwolf-ble-phy-tests)
add_executable(airwolf-ble-connection-tests tests/ble_connection_test.cpp)
target_link_libraries(airwolf-ble-connection-tests PRIVATE airwolf_dsp Threads::Threads)
add_test(NAME native-ble-connection COMMAND airwolf-ble-connection-tests)
add_executable(airwolf-enocean-tests tests/enocean_test.cpp)
target_link_libraries(airwolf-enocean-tests PRIVATE airwolf_dsp Threads::Threads)
add_test(NAME native-enocean COMMAND airwolf-enocean-tests)
add_executable(airwolf-enocean2-tests tests/enocean2_test.cpp)
target_link_libraries(airwolf-enocean2-tests PRIVATE airwolf_dsp Threads::Threads)
add_test(NAME native-enocean2 COMMAND airwolf-enocean2-tests)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(AIRWOLF_WITH_HACKRF AND Python3_Interpreter_FOUND)
  add_test(NAME native-stream COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/stream_test.py" "$<TARGET_FILE:airwolf-radio>")
  set_tests_properties(native-stream PROPERTIES TIMEOUT 30)
endif()
add_executable(airwolf-ble-follow-tests tests/ble_follow_test.cpp)
target_link_libraries(airwolf-ble-follow-tests PRIVATE airwolf_dsp Threads::Threads)
add_test(NAME native-ble-follow COMMAND airwolf-ble-follow-tests)
# Compile the production controller against a fake USB device, without linking
# libhackrf. This exercises actual stop/tune/start, worker and JSON paths on CI.
add_executable(airwolf-follow-transport-tests src/main.cpp tests/fake/hackrf.cpp)
target_include_directories(airwolf-follow-transport-tests BEFORE PRIVATE tests/fake)
target_link_libraries(airwolf-follow-transport-tests PRIVATE airwolf_dsp Threads::Threads)
if(Python3_Interpreter_FOUND)
  add_test(NAME native-follow-transport COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/follow_stream_test.py" "$<TARGET_FILE:airwolf-follow-transport-tests>")
  set_tests_properties(native-follow-transport PROPERTIES TIMEOUT 20)
endif()
