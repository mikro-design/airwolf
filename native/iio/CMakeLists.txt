cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(airwolf_iio LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  enable_testing()
  find_package(Threads REQUIRED)
  add_library(airwolf_dsp INTERFACE)
  target_include_directories(airwolf_dsp INTERFACE ../src)
  if(APPLE)
    find_library(ACCELERATE Accelerate REQUIRED)
    target_link_libraries(airwolf_dsp INTERFACE "${ACCELERATE}")
    target_compile_definitions(airwolf_dsp INTERFACE AIRWOLF_ACCELERATE)
  endif()
endif()
list(PREPEND CMAKE_PREFIX_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../iio-deps/install")
find_package(PkgConfig REQUIRED)
pkg_check_modules(IIO REQUIRED libiio>=1.0)
# libiio 1.x names its Windows library iio1, but upstream's .pc still says
# -liio. Resolve the actual import library, never the 0.x compatibility DLL.
find_library(AIRWOLF_IIO_LIBRARY NAMES iio1 libiio1 iio HINTS ${IIO_LIBRARY_DIRS} NO_DEFAULT_PATH REQUIRED)
add_library(airwolf_libiio UNKNOWN IMPORTED)
set_target_properties(airwolf_libiio PROPERTIES IMPORTED_LOCATION "${AIRWOLF_IIO_LIBRARY}"
  INTERFACE_INCLUDE_DIRECTORIES "${IIO_INCLUDE_DIRS}")
add_library(airwolf_iio_source STATIC source.cpp)
target_link_libraries(airwolf_iio_source PUBLIC airwolf_libiio)
target_include_directories(airwolf_iio_source PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(airwolf-iio main.cpp live.cpp)
target_link_libraries(airwolf-iio PRIVATE airwolf_iio_source airwolf_dsp Threads::Threads)
set_target_properties(airwolf-iio PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(airwolf-iio PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
endforeach()
install(TARGETS airwolf-iio RUNTIME DESTINATION bin)
add_executable(airwolf-iio-format-tests tests/format.cpp)
target_link_libraries(airwolf-iio-format-tests PRIVATE airwolf_dsp)
target_include_directories(airwolf-iio-format-tests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME native-iio-format COMMAND airwolf-iio-format-tests)

foreach(protocol gps lte lora)
  if(TARGET airwolf_${protocol}_receiver)
    string(TOUPPER "${protocol}" definition)
    target_compile_definitions(airwolf-iio PRIVATE AIRWOLF_${definition})
    target_link_libraries(airwolf-iio PRIVATE airwolf_${protocol}_receiver)
  endif()
endforeach()

add_executable(airwolf-iio-transport-tests main.cpp live.cpp tests/fake_source.cpp)
target_link_libraries(airwolf-iio-transport-tests PRIVATE airwolf_dsp Threads::Threads)
foreach(protocol gps lte lora)
  if(TARGET airwolf_${protocol}_receiver)
    string(TOUPPER "${protocol}" definition)
    target_compile_definitions(airwolf-iio-transport-tests PRIVATE AIRWOLF_${definition})
    target_link_libraries(airwolf-iio-transport-tests PRIVATE airwolf_${protocol}_receiver)
  endif()
endforeach()

if(WIN32)
  # Make developer/test binaries runnable before installer staging.
  find_file(AIRWOLF_IIO_DLL NAMES libiio1.dll iio1.dll
    PATHS "${CMAKE_CURRENT_SOURCE_DIR}/../iio-deps/install/bin" NO_DEFAULT_PATH REQUIRED)
  add_custom_command(TARGET airwolf-iio POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${AIRWOLF_IIO_DLL}" "$<TARGET_FILE_DIR:airwolf-iio>")
endif()
if(AIRWOLF_SIGNAL_DEPS EQUAL 0)
  add_test(NAME native-iio-transport COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/transport.py"
    "$<TARGET_FILE:airwolf-iio-transport-tests>" "$<TARGET_FILE:airwolf-iq>")
  set_tests_properties(native-iio-transport PROPERTIES TIMEOUT 120)
endif()
if(AIRWOLF_SIGNAL_DEPS EQUAL 0 AND TARGET airwolf-lte-radio AND TARGET airwolf-lora-radio AND TARGET airwolf-gps-radio)
  add_test(NAME native-iio-protocol-replay COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/replay.py" "${CMAKE_BINARY_DIR}")
  set_tests_properties(native-iio-protocol-replay PROPERTIES TIMEOUT 180)
endif()
