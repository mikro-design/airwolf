enable_language(C)
find_package(PkgConfig REQUIRED)
set(AIRWOLF_GPS_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../gps-deps/PocketSDR-23c91c732365a3668d303656de95a36e7604b58a" CACHE PATH "Pinned PocketSDR receive source")
if(NOT EXISTS "${AIRWOLF_GPS_SOURCE}/src/sdr_ch.c")
  message(FATAL_ERROR "Run node scripts/prepare-gps.mjs before enabling GPS reception")
endif()
pkg_check_modules(GPS_FFTW REQUIRED IMPORTED_TARGET fftw3f)
pkg_check_modules(GPS_USB REQUIRED IMPORTED_TARGET libusb-1.0)
set(gps_sources)
foreach(file sdr_cmn sdr_func sdr_code sdr_code_gal sdr_ch sdr_nav sdr_fec sdr_ldpc sdr_nb_ldpc)
  list(APPEND gps_sources "${AIRWOLF_GPS_SOURCE}/src/${file}.c")
endforeach()
foreach(file rtkcmn tides rtkpos geoid solution lambda sbas stream rcvraw rtcm rtcm2 rtcm3 rtcm3e preceph options pntpos ppp ppp_ar ephemeris rinex ionex convrnx streamsvr rcv/binex rcv/ublox rcv/novatel rcv/septentrio)
  list(APPEND gps_sources "${AIRWOLF_GPS_SOURCE}/lib/RTKLIB/src/${file}.c")
endforeach()
add_library(airwolf_gps_core STATIC ${gps_sources})
target_include_directories(airwolf_gps_core PUBLIC "${AIRWOLF_GPS_SOURCE}/src" "${AIRWOLF_GPS_SOURCE}/lib/RTKLIB/src")
target_link_libraries(airwolf_gps_core PUBLIC PkgConfig::GPS_FFTW PkgConfig::GPS_USB Threads::Threads)
target_compile_definitions(airwolf_gps_core PRIVATE FFTW AIRWOLF_GPS)
if(APPLE)
  target_compile_definitions(airwolf_gps_core PRIVATE MACOS)
endif()
if(WIN32)
  target_compile_definitions(airwolf_gps_core PUBLIC WIN32)
  target_link_libraries(airwolf_gps_core PUBLIC ws2_32 winmm)
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64|ARM64)$")
  target_compile_definitions(airwolf_gps_core PRIVATE NEON)
endif()
if(NOT MSVC)
  target_compile_options(airwolf_gps_core PRIVATE -w)
endif()
add_library(airwolf_gps_receiver STATIC receiver.cpp)
target_link_libraries(airwolf_gps_receiver PUBLIC airwolf_dsp PRIVATE airwolf_gps_core)
target_include_directories(airwolf_gps_receiver PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(AIRWOLF_WITH_HACKRF)
  add_executable(airwolf-gps-radio ../src/main.cpp ../src/iq_decode.cpp)
  target_compile_definitions(airwolf-gps-radio PRIVATE AIRWOLF_GPS)
  target_link_libraries(airwolf-gps-radio PRIVATE airwolf_gps_receiver ${HACKRF_TARGET} Threads::Threads)
  if(WIN32)
    target_link_libraries(airwolf-gps-radio PRIVATE winmm)
  endif()
  set_target_properties(airwolf-gps-radio PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
    set_target_properties(airwolf-gps-radio PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
  endforeach()
endif()
add_executable(airwolf-gps-transport-tests ../src/main.cpp ../src/iq_decode.cpp ../tests/fake/hackrf.cpp)
target_compile_definitions(airwolf-gps-transport-tests PRIVATE AIRWOLF_GPS AIRWOLF_TEST_CLOCK)
target_include_directories(airwolf-gps-transport-tests BEFORE PRIVATE ../tests/fake)
target_link_libraries(airwolf-gps-transport-tests PRIVATE airwolf_gps_receiver Threads::Threads)
if(WIN32)
  target_link_libraries(airwolf-gps-transport-tests PRIVATE winmm)
endif()
add_executable(airwolf-gps-feed tests/feed.cpp)
target_link_libraries(airwolf-gps-feed PRIVATE airwolf_gps_receiver airwolf_gps_core)
add_executable(airwolf-gps-lnav-tests tests/lnav.cpp)
target_link_libraries(airwolf-gps-lnav-tests PRIVATE airwolf_dsp)
if(Python3_Interpreter_FOUND)
  add_test(NAME native-gps-navigation COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/navigation.py" "$<TARGET_FILE:airwolf-gps-lnav-tests>")
  set_tests_properties(native-gps-navigation PROPERTIES TIMEOUT 60)
endif()
if(Python3_Interpreter_FOUND AND AIRWOLF_WITH_HACKRF)
  add_test(NAME native-gps-reception COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/receive.py" "$<TARGET_FILE:airwolf-gps-feed>" "$<TARGET_FILE:airwolf-gps-radio>")
  set_tests_properties(native-gps-reception PROPERTIES TIMEOUT 240)
endif()
