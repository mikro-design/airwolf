enable_language(C)
find_package(PkgConfig REQUIRED)
set(AIRWOLF_GPS_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../gps-deps/PocketSDR-23c91c732365a3668d303656de95a36e7604b58a" CACHE PATH "Pinned PocketSDR receive source")
if(NOT EXISTS "${AIRWOLF_GPS_SOURCE}/src/sdr_ch.c")
  message(FATAL_ERROR "Run node scripts/prepare-gps.mjs before enabling GPS reception")
endif()
pkg_check_modules(GPS_FFTW REQUIRED IMPORTED_TARGET fftw3f)
pkg_check_modules(GPS_USB REQUIRED IMPORTED_TARGET libusb-1.0)
set(gps_sources)
foreach(file sdr_cmn sdr_func sdr_code sdr_code_gal sdr_ch sdr_nav sdr_fec sdr_ldpc sdr_nb_ldpc)
  list(APPEND gps_sources "${AIRWOLF_GPS_SOURCE}/src/${file}.c")
endforeach()
foreach(file rtkcmn tides rtkpos geoid solution lambda sbas stream rcvraw rtcm rtcm2 rtcm3 rtcm3e preceph options pntpos ppp ppp_ar ephemeris rinex ionex convrnx streamsvr rcv/binex rcv/ublox rcv/novatel rcv/septentrio)
  list(APPEND gps_sources "${AIRWOLF_GPS_SOURCE}/lib/RTKLIB/src/${file}.c")
endforeach()
if(APPLE)
  # vDSP takes PocketSDR's FFTs (gps_vdsp.c): rename the FFTW implementation
  # in a copy of sdr_func.c so it remains the fallback for other lengths.
  file(READ "${AIRWOLF_GPS_SOURCE}/src/sdr_func.c" airwolf_sdr_func)
  set(airwolf_fft_old "int sdr_cpx_fft(sdr_cpx_t *cpx1, int N, int dir, sdr_cpx_t *cpx2)\n{")
  string(FIND "${airwolf_sdr_func}" "${airwolf_fft_old}" airwolf_fft_at)
  if(airwolf_fft_at EQUAL -1)
    message(FATAL_ERROR "Pinned PocketSDR sdr_cpx_fft no longer matches the vDSP replacement")
  endif()
  string(REPLACE "${airwolf_fft_old}" "int airwolf_fftw_cpx_fft(sdr_cpx_t *cpx1, int N, int dir, sdr_cpx_t *cpx2)\n{" airwolf_sdr_func "${airwolf_sdr_func}")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/sdr_func_vdsp.c" "${airwolf_sdr_func}")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${AIRWOLF_GPS_SOURCE}/src/sdr_func.c")
  list(REMOVE_ITEM gps_sources "${AIRWOLF_GPS_SOURCE}/src/sdr_func.c")
  list(APPEND gps_sources "${CMAKE_CURRENT_BINARY_DIR}/sdr_func_vdsp.c" "${CMAKE_CURRENT_SOURCE_DIR}/gps_vdsp.c")
endif()
add_library(airwolf_gps_core STATIC ${gps_sources})
target_include_directories(airwolf_gps_core PUBLIC "${AIRWOLF_GPS_SOURCE}/src" "${AIRWOLF_GPS_SOURCE}/lib/RTKLIB/src")
target_link_libraries(airwolf_gps_core PUBLIC PkgConfig::GPS_FFTW PkgConfig::GPS_USB Threads::Threads)
target_compile_definitions(airwolf_gps_core PRIVATE FFTW AIRWOLF_GPS decode_frame=airwolf_gps_decode_frame)
if(APPLE)
  target_compile_definitions(airwolf_gps_core PRIVATE MACOS)
  find_library(AIRWOLF_GPS_ACCELERATE Accelerate REQUIRED)
  target_link_libraries(airwolf_gps_core PUBLIC "${AIRWOLF_GPS_ACCELERATE}")
endif()
if(WIN32)
  target_compile_definitions(airwolf_gps_core PUBLIC WIN32)
  target_link_libraries(airwolf_gps_core PUBLIC ws2_32 winmm)
endif()
# x86 builds use PocketSDR's portable C correlators. AIRWOLF_GPS_GENERIC selects
# them on ARM too, to reproduce x86 receiver behaviour on Apple Silicon.
option(AIRWOLF_GPS_GENERIC "Use portable GPS correlators instead of NEON" OFF)
if(NOT AIRWOLF_GPS_GENERIC AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64|ARM64)$")
  target_compile_definitions(airwolf_gps_core PRIVATE NEON)
endif()
if(NOT MSVC)
  target_compile_options(airwolf_gps_core PRIVATE -w)
endif()

target_link_libraries(airwolf_gps_receiver PUBLIC airwolf_dsp PRIVATE airwolf_gps_core)
target_include_directories(airwolf_gps_receiver PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")

  endforeach()
